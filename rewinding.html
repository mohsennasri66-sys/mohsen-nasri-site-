async function loadGallery(){ const gallery = document.getElementById("rewindingGallery"); gallery.innerHTML = `
در حال دریافت تصاویر...
`; try{ const {data,error} = await supabaseClient .from("industry_media") .select("id,industry,media_type,file_path,title") .eq("industry","rewinding") .eq("media_type","image") .order("id",{ascending:false}); console.log("GALLERY DATA:", data); console.log("GALLERY ERROR:", error); if(error){ gallery.innerHTML = `
خطا در دریافت تصاویر:

${escapeHtml(error.message)}
`; return; } if(!data || data.length === 0){ gallery.innerHTML = `
هنوز تصویری برای این صنعت ثبت نشده است.
`; return; } gallery.innerHTML = ""; data.forEach(item => { if(!item.file_path){ return; } const imageUrl = "https://eurmybgotfgnerdxrwua.supabase.co/storage/v1/object/public/project-media/" + item.file_path; console.log("IMAGE URL:", imageUrl); const card = document.createElement("div"); card.className = "gallery-card"; const img = document.createElement("img"); img.src = imageUrl; img.alt = item.title || "تصویر صنعت سیم‌پیچی"; img.loading = "lazy"; img.onclick = function(){ window.open( imageUrl, "_blank" ); }; img.onerror = function(){ console.error( "IMAGE LOAD ERROR:", imageUrl ); card.innerHTML = `
خطا در نمایش تصویر

مسیر:
${escapeHtml(item.file_path)}
`; }; card.appendChild(img); if(item.title){ const title = document.createElement("div"); title.className = "gallery-title"; title.textContent = item.title; card.appendChild(title); } gallery.appendChild(card); }); if(!gallery.children.length){ gallery.innerHTML = `
رکورد تصویر وجود دارد اما مسیر فایل خالی است.
`; } }catch(error){ console.error( "GALLERY ERROR:", error ); gallery.innerHTML = `
خطا در اتصال به گالری:

${escapeHtml(error.message)}
`; } }
